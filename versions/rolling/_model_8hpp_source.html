698struct JointTypeVisitor : public boost::static_visitor<JointType> {
699 // revolute (bounded)
700 JointType operator()(const pinocchio::JointModelRevoluteTpl<double, 0, 0>&) const { return JointType::REVOLUTE; }
701 JointType operator()(const pinocchio::JointModelRevoluteTpl<double, 0, 1>&) const { return JointType::REVOLUTE; }
702 JointType operator()(const pinocchio::JointModelRevoluteTpl<double, 0, 2>&) const { return JointType::REVOLUTE; }
703
704 // continuous (unbounded revolute)
705 JointType operator()(const pinocchio::JointModelRevoluteUnboundedTpl<double, 0, 0>&) const {
706 return JointType::CONTINUOUS;
707 }
708 JointType operator()(const pinocchio::JointModelRevoluteUnboundedTpl<double, 0, 1>&) const {
709 return JointType::CONTINUOUS;
710 }
711 JointType operator()(const pinocchio::JointModelRevoluteUnboundedTpl<double, 0, 2>&) const {
712 return JointType::CONTINUOUS;
713 }
714
715 // prismatic
716 JointType operator()(const pinocchio::JointModelPrismaticTpl<double, 0, 0>&) const { return JointType::PRISMATIC; }
717 JointType operator()(const pinocchio::JointModelPrismaticTpl<double, 0, 1>&) const { return JointType::PRISMATIC; }
718 JointType operator()(const pinocchio::JointModelPrismaticTpl<double, 0, 2>&) const { return JointType::PRISMATIC; }
719
720 // planar
721 JointType operator()(const pinocchio::JointModelPlanarTpl<double, 0>&) const { return JointType::PLANAR; }
722
723 // free-flyer
724 JointType operator()(const pinocchio::JointModelFreeFlyerTpl<double, 0>&) const { return JointType::FLOATING; }
725
726 template<typename JointModel>
727 JointType operator()(const JointModel&) const {
728 return JointType::NONE;
729 }
730};

732inline std::vector<JointType> Model::get_joint_types() {
733 std::vector<JointType> joint_types;
734 JointTypeVisitor classifier;
735 for (int j = 1; j < this->robot_model_.njoints; ++j) {// skips the first joint (universe)
736 const auto& jmv = this->robot_model_.joints[j];
737 JointType type = boost::apply_visitor(classifier, jmv);
738 joint_types.push_back(type);
739 }
740 return joint_types;
741}
